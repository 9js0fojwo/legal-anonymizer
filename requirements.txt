# Legal Document Anonymizer - Requirements
# 法律文档脱敏工具 - 依赖包

# Core dependencies (required)
# 核心依赖（必需）
pymupdf>=1.23.0        # PyMuPDF for PDF processing - PDF处理
python-docx>=1.1.0      # Word document processing - Word文档处理
reportlab>=4.0.0        # PDF generation - PDF生成
flask>=3.0.0            # Web UI - 网页界面

# Optional dependencies (for advanced features)
# 可选依赖（高级功能）
pillow>=10.0.0          # Image processing for OCR - 图像处理（OCR用）
numpy>=1.21.0           # Array processing for PaddleOCR - 数组处理

# OCR engines (install at least one)
# OCR引擎（至少安装一个）
paddlepaddle>=3.0.0     # PaddlePaddle deep learning framework (recommended) - 飞桨深度学习框架（推荐）
paddleocr>=3.0.0        # PaddleOCR - high accuracy Chinese OCR (recommended) - 高精度中文OCR（推荐）
pytesseract>=0.3.10     # Tesseract OCR fallback - Tesseract OCR备选

# Visual redaction for scanned PDFs (required to preserve original page images)
# 扫描版 PDF 视觉脱敏必装：保留盖章/签名/版式，只覆盖白框/占位符
# 没装这两个的话，扫描版 PDF 会回退到模板渲染（仿宋小四），原版式丢失
rapidocr>=3.8.0         # RapidOCR - returns char-level bbox - 字符级位置框
onnxruntime>=1.20.0     # RapidOCR ONNX backend - 推理后端

# LLM-based PII detectors (optional)
# 可选：补充检测层，补规则盲区。三层各自独立开启，互不冲突
#
#   --cn-llm : uer/roberta-base-finetuned-cluener2020-chinese (~400MB)
#              补规则漏掉的中文人名/公司/地址（含复姓）
#              pip install torch transformers
#
#   --llm    : openai/privacy-filter 1.5B MoE (~2.6GB)
#              补英文人名/地址/API token/国际电话
#              pip install torch transformers "httpx[socks]"
#
#   --ollama : 本地部署的大模型（Qwen/Gemma 等），通过 Ollama API 调用
#              无需额外安装 Python 依赖；需本机运行 Ollama（https://ollama.com）
#              默认连接 http://localhost:11434，模型 qwen2.5:7b
#              远端服务器（如 Tailscale 内网）：
#                --ollama-url http://100.x.x.x:11434 --ollama-model qwen2.5:7b
#              环境变量方式：
#                LEGAL_ANONYMIZER_OLLAMA=1
#                LEGAL_ANONYMIZER_OLLAMA_URL=http://100.x.x.x:11434
#                LEGAL_ANONYMIZER_OLLAMA_MODEL=qwen2.5:7b
#
# --cn-llm 和 --llm 首次启用会从 HuggingFace 下载到 ~/.cache/huggingface，之后离线推理
# transformers>=4.40.0
# torch>=2.0.0

# Installation instructions:
# 安装说明：
#
# Basic installation (minimal):
# pip install pymupdf python-docx reportlab flask
#
# Recommended OCR installation (PaddleOCR - best for Chinese):
# pip install paddlepaddle paddleocr pillow numpy
#
# Alternative OCR installation (Tesseract):
# pip install pillow pytesseract
# Also install Tesseract itself:
#   macOS: brew install tesseract
#   Ubuntu: sudo apt install tesseract-ocr
#   Windows: Download from https://github.com/UB-Mannheim/tesseract/wiki
#
# Full installation (all features):
# pip install -r requirements.txt
